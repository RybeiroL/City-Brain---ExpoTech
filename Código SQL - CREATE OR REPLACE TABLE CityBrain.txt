CREATE OR REPLACE TABLE CityBrain.tb_fato_final AS

SELECT
    f.id_fato,
    
    p.data,
    
    l.regiao,
    l.bairro,
    
    i.nome_indicador AS indicador,
    i.categoria,
    
    o.tipo_ocorrencia,
    o.descricao,
    
    f.valor,
    f.ocorrencias

FROM CityBrain.tb_fato f

LEFT JOIN CityBrain.tb_indicador i
    ON f.id_indicador = i.id_indicador

LEFT JOIN CityBrain.tb_localizacao l
    ON f.id_localizacao = l.id_localizacao

LEFT JOIN CityBrain.tb_periodo p
    ON f.id_periodo = p.id_periodo

LEFT JOIN CityBrain.tb_ocorrencia o
    ON f.id_ocorrencia = o.id_ocorrencia;
